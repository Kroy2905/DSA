Pattern 1: Two Pointers

P1-1Two Sum — Find pair that sums to targetEasy 
// O(n) time, O(n) space — HashMap solution
fun twoSum(nums: IntArray, target: Int): IntArray {
    val seen = HashMap<Int, Int>() // value → index
    for (i in nums.indices) {
        val complement = target - nums[i]
        if (seen.containsKey(complement)) {
            return intArrayOf(seen[complement]!!, i)
        }
        seen[nums[i]] = i
    }
    return intArrayOf()
}
// Key insight: complement = target - current. Check if complement seen before.
P1-2Container With Most Water (Two Pointers)Medium 
// Greedy two-pointer: move the shorter side inward
fun maxArea(height: IntArray): Int {
    var left = 0; var right = height.size - 1; var maxWater = 0
    while (left < right) {
        val water = minOf(height[left], height[right]) * (right - left)
        maxWater = maxOf(maxWater, water)
        if (height[left] < height[right]) left++ else right--
        // Move the shorter side — the taller side can't improve by moving
    }
    return maxWater
}
// O(n) time, O(1) space

Pattern 2: Sliding Window

P2-1Longest Substring Without Repeating CharactersMedium 
fun lengthOfLongestSubstring(s: String): Int {
    val lastSeen = HashMap<Char, Int>() // char → last index seen
    var left = 0; var maxLen = 0
    for (right in s.indices) {
        if (lastSeen.containsKey(s[right])) {
            // Shrink window past the duplicate
            left = maxOf(left, lastSeen[s[right]]!! + 1)
        }
        lastSeen[s[right]] = right
        maxLen = maxOf(maxLen, right - left + 1)
    }
    return maxLen
}
// O(n) time, O(min(n,m)) space where m = charset size
P2-2Minimum Window SubstringHard 
fun minWindow(s: String, t: String): String {
    if (s.isEmpty() || t.isEmpty()) return ""
    val need = HashMap<Char, Int>()
    for (c in t) need[c] = (need[c] ?: 0) + 1
    var left = 0; var formed = 0; val window = HashMap<Char, Int>()
    var minLen = Int.MAX_VALUE; var minLeft = 0
    for (right in s.indices) {
        window[s[right]] = (window[s[right]] ?: 0) + 1
        if (need.containsKey(s[right]) && window[s[right]] == need[s[right]]) formed++
        while (formed == need.size) { // window is valid, try to shrink
            if (right - left + 1 < minLen) { minLen = right - left + 1; minLeft = left }
            window[s[left]] = window[s[left]]!! - 1
            if (need.containsKey(s[left]) && window[s[left]]!! < need[s[left]]!!) formed--
            left++
        }
    }
    return if (minLen == Int.MAX_VALUE) "" else s.substring(minLeft, minLeft + minLen)
}

Pattern 3: Binary Search

P3-1Classic binary search template — know this coldEasy 
// Template: find exact target
fun binarySearch(nums: IntArray, target: Int): Int {
    var lo = 0; var hi = nums.size - 1
    while (lo <= hi) {
        val mid = lo + (hi - lo) / 2 // avoid overflow
        when {
            nums[mid] == target -> return mid
            nums[mid] < target -> lo = mid + 1
            else -> hi = mid - 1
        }
    }
    return -1
}

// Template: find leftmost position (first occurrence)
fun lowerBound(nums: IntArray, target: Int): Int {
    var lo = 0; var hi = nums.size // hi = size (not size-1)
    while (lo < hi) { // strict less (not <=)
        val mid = lo + (hi - lo) / 2
        if (nums[mid] < target) lo = mid + 1 else hi = mid
    }
    return lo // first position where nums[pos] >= target
}
P3-2Search in Rotated Sorted ArrayMedium — Google Favorite 
fun search(nums: IntArray, target: Int): Int {
    var lo = 0; var hi = nums.size - 1
    while (lo <= hi) {
        val mid = lo + (hi - lo) / 2
        if (nums[mid] == target) return mid
        // Determine which half is sorted
        if (nums[lo] <= nums[mid]) { // left half is sorted
            if (target >= nums[lo] && target < nums[mid]) hi = mid - 1
            else lo = mid + 1
        } else { // right half is sorted
            if (target > nums[mid] && target <= nums[hi]) lo = mid + 1
            else hi = mid - 1
        }
    }
    return -1
}
// Key insight: at least one half is always sorted. Check which half, then decide.

Pattern 4: Trees

P4-1BFS and DFS templates for treesEasy 
class TreeNode(var value: Int, var left: TreeNode? = null, var right: TreeNode? = null)

// DFS: inorder (left-root-right) — gives sorted order for BST
fun inorder(root: TreeNode?, result: MutableList<Int>) {
    if (root == null) return
    inorder(root.left, result)
    result.add(root.value)
    inorder(root.right, result)
}

// BFS: level order traversal
fun levelOrder(root: TreeNode?): List<List<Int>> {
    if (root == null) return emptyList()
    val result = mutableListOf<List<Int>>()
    val queue: ArrayDeque<TreeNode> = ArrayDeque()
    queue.addLast(root)
    while (queue.isNotEmpty()) {
        val levelSize = queue.size
        val level = mutableListOf<Int>()
        repeat(levelSize) {
            val node = queue.removeFirst()
            level.add(node.value)
            node.left?.let { queue.addLast(it) }
            node.right?.let { queue.addLast(it) }
        }
        result.add(level)
    }
    return result
}
P4-2Lowest Common Ancestor of Binary TreeMedium — Google Favorite 
fun lowestCommonAncestor(root: TreeNode?, p: TreeNode?, q: TreeNode?): TreeNode? {
    if (root == null || root == p || root == q) return root
    val left = lowestCommonAncestor(root.left, p, q)
    val right = lowestCommonAncestor(root.right, p, q)
    // If found in both subtrees, root is LCA
    if (left != null && right != null) return root
    return left ?: right
}
// O(n) time — visits each node once. Key: if both subtrees return non-null, root is LCA

Pattern 5: Dynamic Programming

P5-1Climbing Stairs — DP foundationEasy 
// Recurrence: ways(n) = ways(n-1) + ways(n-2)
// O(n) time, O(1) space — rolling variables
fun climbStairs(n: Int): Int {
    if (n <= 2) return n
    var prev2 = 1; var prev1 = 2
    for (i in 3..n) {
        val cur = prev1 + prev2
        prev2 = prev1; prev1 = cur
    }
    return prev1
}
P5-2Longest Common SubsequenceMedium 
fun longestCommonSubsequence(text1: String, text2: String): Int {
    val m = text1.length; val n = text2.length
    val dp = Array(m + 1) { IntArray(n + 1) } // dp[i][j] = LCS of text1[0..i-1] and text2[0..j-1]
    for (i in 1..m) {
        for (j in 1..n) {
            dp[i][j] = if (text1[i-1] == text2[j-1]) dp[i-1][j-1] + 1
                       else maxOf(dp[i-1][j], dp[i][j-1])
        }
    }
    return dp[m][n]
}
// O(m*n) time and space. Classic 2D DP problem.

Pattern 6: Graph BFS/DFS

P6-1Number of Islands — DFS on gridMedium — Must Know 
fun numIslands(grid: Array<CharArray>): Int {
    var count = 0
    for (r in grid.indices) {
        for (c in grid[r].indices) {
            if (grid[r][c] == '1') {
                count++
                dfs(grid, r, c) // mark all connected land as visited
            }
        }
    }
    return count
}

private fun dfs(grid: Array<CharArray>, r: Int, c: Int) {
    if (r < 0 || r >= grid.size || c < 0 || c >= grid[0].size || grid[r][c] != '1') return
    grid[r][c] = '0' // mark visited
    dfs(grid, r+1, c); dfs(grid, r-1, c)
    dfs(grid, r, c+1); dfs(grid, r, c-1)
}
// O(m*n) time. Modify grid in-place to avoid visited set.
P6-2Word Ladder — BFS shortest pathHard 
fun ladderLength(beginWord: String, endWord: String, wordList: List<String>): Int {
    val wordSet = wordList.toHashSet()
    if (!wordSet.contains(endWord)) return 0
    val queue: ArrayDeque<String> = ArrayDeque()
    queue.addLast(beginWord); var steps = 1
    while (queue.isNotEmpty()) {
        repeat(queue.size) {
            val word = queue.removeFirst()
            for (i in word.indices) {
                for (c in 'a'..'z') {
                    val next = word.substring(0, i) + c + word.substring(i+1)
                    if (next == endWord) return steps + 1
                    if (wordSet.remove(next)) queue.addLast(next)
                }
            }
        }
        steps++
    }
    return 0
}
// BFS guarantees shortest path. Remove from set to mark visited.

Interview Tips for DSA

★How to approach a coding problem at GoogleMust Know
  1. Clarify (2 min): Edge cases? Input constraints? Can I modify input? Return type?
  2. Example (2 min): Walk through 1-2 examples manually. Find the pattern.
  3. Brute force (2 min): State the O(n²) or obvious solution. "Naive approach is X."
  4. Optimize (5 min): Think about: sorting, two pointers, sliding window, hash map, DP, divide and conquer. State before coding.
  5. Code (15 min): Clean, readable code. Name variables well. Comments on non-obvious parts.
  6. Test (5 min): Walk through your code with the example. Test edge cases: empty, single element, all same.
  7. Complexity: State time and space complexity. Know how to derive it.
✓ Never code in silence. Think out loud. Interviewers want to see your process, not just the answer.
💡 Most Google questions use: HashMap, BFS/DFS, Two Pointers, Binary Search, or DP. Recognize the pattern first.
